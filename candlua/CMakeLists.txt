cmake_minimum_required(VERSION 4.0)
set(CMAKE_C_STANDARD 23)
project(candlua C)

find_package(Lua 5.4 QUIET)
add_executable(luadyn ref_lua_demo.c dynref.c)
add_executable(event_loop event_loop.c)

if(TARGET Lua::Lua)
	target_link_libraries(luatest PRIVATE Lua::Lua m dl)
	target_link_libraries(luadyn PRIVATE Lua::Lua m dl)
	target_link_libraries(event_loop PRIVATE Lua::Lua m dl)
elseif(Lua_FOUND)
	if(NOT DEFINED LUA_INCLUDE_DIR AND DEFINED LUA_INCLUDES)
		set(LUA_INCLUDE_DIR ${LUA_INCLUDES})
	endif()
	target_include_directories(luadyn PRIVATE ${LUA_INCLUDE_DIR})
	target_link_libraries(luadyn PRIVATE ${LUA_LIBRARIES} m dl)
	target_include_directories(event_loop PRIVATE ${LUA_INCLUDE_DIR})
	target_link_libraries(event_loop PRIVATE ${LUA_LIBRARIES} m dl)
else()
	find_package(PkgConfig REQUIRED)
	pkg_check_modules(LUA_PKG QUIET lua5.4 lua54 lua)
	if(NOT LUA_PKG_FOUND)
		message(FATAL_ERROR "Could not find Lua. Install Lua dev package or adjust pkg-config module name (e.g. lua5.4).")
	endif()

	target_include_directories(luadyn PRIVATE ${LUA_PKG_INCLUDE_DIRS})
	target_link_libraries(luadyn PRIVATE ${LUA_PKG_LIBRARIES} ${LUA_PKG_LDFLAGS} m dl)
	target_compile_options(luadyn PRIVATE ${LUA_PKG_CFLAGS_OTHER})

	target_include_directories(event_loop PRIVATE ${LUA_PKG_INCLUDE_DIRS})
	target_link_libraries(event_loop PRIVATE ${LUA_PKG_LIBRARIES} ${LUA_PKG_LDFLAGS} m dl)
	target_compile_options(event_loop PRIVATE ${LUA_PKG_CFLAGS_OTHER})
endif()

message(STATUS "Building targets: luatest luadyn event_loop")
